    if (n === 0) return 1; // handling the scenario if the number is 0 and we want the count to be 1
    // if (n < 0) n *= -1 // handling case of negative number
    n = Math.abs(n); // handling for case of negative number
    let count = 0;
    while (n > 0) {
        n = Math.floor(n / 10);

function checkPalindrome(n) {
    if(n<0) return false; // If negative numbers must be reversed with sign os it wont be a palindrome
    if(n>=0 && n<=9) return false; // If single digit is taken as not a palindrome
    const originalNum = n;
    let rev = 0;

    while(n>0) {
        const rem = n%10;
        rev = (10*rev) + rem;
        n= Math.abs(n/10);
    }
    return rev === originalNum;
}


// REVERSE AN INTEGER (SIGNED INTEGER WITHIN THE LIMIT [-2^31, (2^31)-1]);

function reverseInteger(n) {
    const min = -(2 ** 31); // Math.pow(2,31)
    const max = (2 ** 31) - 1; // Math.pow(2,31) - 1

    let revNum = n < 0 ? "-" : "";
    n = Math.abs(n);
    while (n > 0) {
        const rem = n % 10;
        revNum += rem;
        n = Math.floor(n / 10);
    }
    if (Number(revNum) < min || Number(revNum) > max) return 0;
    return (Math.floor(revNum));
}
