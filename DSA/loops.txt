const arr = [2,3,5,7,12,34,56];
const numberToFind = 34;

function findIndex (arr) {
    let index = -1;
    for(let i=0; i<arr.length; i++){
        if(arr[i] === numberToFind) {
            index = i;
        }
    }
    return index;
}

    let count = 0;

    let secondLargest = arr[0];
    for(let i = 0; i< arr.length; i++) {
        if(arr[i] > secondLargest) {

function findSecondLargest(arr) {
    if(arr.length < 2) return null; // handling the corner cases 1 and 2
    let largest = arr[0];
    let secondLargest = arr[0];
    for(let i = 0; i< arr.length; i++) {
        if(arr[i] > largest) {
            secondLargest = largest;
            largest = arr[i];
            }
        if(arr[i] > secondLargest && arr[i] !== largest) { // checking if my current element is greater than secondLargest and not equal to largest
            secondLargest = arr[i];
        }
    }
    return secondLargest;
}

function printPattern(n) {

        let row = "";

            row+= " ";
        }
